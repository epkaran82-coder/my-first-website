function showMessage() { document.getElementById("message").textContent = "Thank you for visiting our hospital website!"; } const form = document.getElementById("chatForm"); const question = document.getElementById("question"); const messages = document.getElementById("chatMessages"); const sendButton = document.getElementById("sendButton"); const status = document.getElementById("chatStatus"); function addMessage(text, sender) { const item = document.createElement("div"); item.className = "chat-message " + (sender === "user" ? "user-message" : "assistant-message"); item.textContent = text; messages.appendChild(item); messages.scrollTop = messages.scrollHeight; } function getPreparedAnswer(questionText) { const q = questionText.toLowerCase().trim(); if (/^(hi|hello|hey|good morning|good afternoon|good evening)[!. ]*$/.test(q)) { return "Hello! Welcome to District General Hospital Mannar. How can I help you?"; } if (q.includes("thank")) { return "You're welcome! Thank you for visiting our hospital website."; } if (q.includes("help")) { return "You can ask about hospital services, OPD, specialist clinics, emergency care or contact information."; } if (q.includes("service") || q.includes("what do you provide")) { return "Our website lists OPD, inpatient services, specialist clinics and emergency care. Please contact the hospital to confirm specific services and availability."; } if (q.includes("opd") || q.includes("outpatient")) { return "The Outpatient Department provides assessment for patients who do not require admission. Please contact the hospital to confirm current OPD hours."; } if (q.includes("clinic") || q.includes("specialist")) { return "Our website lists specialist clinics. Please contact the hospital to confirm clinic schedules and appointment arrangements."; } if (q.includes("emergency") || q.includes("urgent")) { return "For a life-threatening emergency, seek emergency medical care immediately. Please contact local emergency services or attend the nearest appropriate emergency department."; } if (q.includes("doctor") || q.includes("staff") || q.includes("available")) { return "I don't have live information about doctor numbers or availability. Please contact the hospital directly for the latest information."; } if (q.includes("contact") || q.includes("phone") || q.includes("telephone") || q.includes("address") || q.includes("location")) { return "District General Hospital Mannar is located in Mannar, Sri Lanka. Please consult an official hospital source for verified contact details."; } return "Sorry, I don't have a prepared answer for that question yet. Please try asking about hospital services, OPD, specialist clinics, emergency care or contact information."; } form.addEventListener("submit", function(event) { event.preventDefault(); const originalQuestion = question.value.trim(); if (!originalQuestion) return; addMessage(originalQuestion, "user"); question.value = ""; sendButton.disabled = true; question.disabled = true; status.textContent = "Finding a prepared answer..."; const answer = getPreparedAnswer(originalQuestion); addMessage(answer, "assistant"); status.textContent = "Answer provided from prepared information."; sendButton.disabled = false; question.disabled = false; question.focus(); });
